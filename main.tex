\chapter{稳态等效模型}

前文主要通过分开关状态列方程，求解变换器的稳态量。加入器件损耗等因素后，直接求解会更加繁琐。因此，本章建立稳态等效电路模型，以便分析变换器的输入输出关系、变换比及效率。

\section{理想稳态模型}

理想稳态及小纹波前提下，输入功率与输出功率相等：

\vspace{-0.7cm}
\begin{equation}
  P_{\mathrm{in}}=P_{\mathrm{out}}
  \quad\Rightarrow\quad
  V_{\mathrm{in}}I_{\mathrm{in}}=V_{\mathrm{out}}I_{\mathrm{out}}.
  \label{equ0201}
\end{equation}
\vspace{-0.7cm}

\noindent{}使用直流电压变比 $M(D)$ 表示，则其中满足

\vspace{-0.8cm}
\begin{align}
  V_{\mathrm{out}} & =M(D)V_{\mathrm{in}},
  \label{equ0202}                           \\
  I_{\mathrm{in}}  & =M(D)I_{\mathrm{out}}.
  \label{equ0203}
\end{align}
\vspace{-1cm}

\subsection{直流“变压器”模型}

上述端口关系也可以用变比为 $1:M(D)$ 、其他均为理想参数的直流“变压器”表示，$D$ 为控制输入。

若原边源电压记作 $V_{\mathrm{in}}$，原边等效内阻为 $R_1$，副边负载为 $R$，将原边折算到副边后，电压按变比缩放，电阻按变比的平方缩放：
\begin{equation}
  V_{\mathrm{out}}
  =M(D)V_{\mathrm{in}}\frac{R}{R+M^2(D)R_1}.
  \label{equ0204}
\end{equation}

\examplesubheading{小纹波近似}

接状态 1 时，
\begin{align}
   & \left\{
  \begin{aligned}
    v_{\mathrm{L}}(t) & =V_{\mathrm{in}}-i_{\mathrm{L}}(t)R_{\mathrm{L}}
    \approx V_{\mathrm{in}}-I_{\mathrm{L}}R_{\mathrm{L}},                \\
    i_{\mathrm{C}}(t) & =-\frac{v_{\mathrm{out}}(t)}{R}
    \approx-\frac{V_{\mathrm{out}}}{R}.
  \end{aligned}
  \right.
  \label{equ0205} \\
  \intertext{\qquad{}接状态 2 时，}
   & \left\{
  \begin{aligned}
    v_{\mathrm{L}}(t) & =V_{\mathrm{in}}-i_{\mathrm{L}}(t)R_{\mathrm{L}}-v_{\mathrm{out}}(t)
    \approx V_{\mathrm{in}}-I_{\mathrm{L}}R_{\mathrm{L}}-V_{\mathrm{out}},                   \\
    i_{\mathrm{C}}(t) & =i_{\mathrm{L}}(t)-\frac{v_{\mathrm{out}}(t)}{R}
    \approx I_{\mathrm{L}}-\frac{V_{\mathrm{out}}}{R}.
  \end{aligned}
  \right.
  \label{equ0206}
\end{align}

\examplesubheading{变换比与平均状态量}

由伏秒平衡，并与下述电荷平衡关系联立，
\begin{alignat}{2}
  \phantom{\Rightarrow}
   & \;                                                        & \frac{1}{T_{\mathrm{s}}}\int_0^{T_{\mathrm{s}}}v_{\mathrm{L}}(t)\,\mathrm{d}t
   & \approx D(V_{\mathrm{in}}-I_{\mathrm{L}}R_{\mathrm{L}})
  +D'(V_{\mathrm{in}}-I_{\mathrm{L}}R_{\mathrm{L}}-V_{\mathrm{out}})=0,
  \label{equ0207}                                                                                                                              \\
  \Rightarrow
   & \;                                                        & V_{\mathrm{out}}
   & \approx\frac{D'R}{R(D')^2+R_{\mathrm{L}}}V_{\mathrm{in}},
  \label{equ0208}                                                                                                                              \\
  \Rightarrow
   & \;                                                        & M(D)
   & =\frac{V_{\mathrm{out}}}{V_{\mathrm{in}}}
  \approx\frac{1}{D'\left(1+\dfrac{R_{\mathrm{L}}}{R(D')^2}\right)}.
  \label{equ0209}                                                                                                                              \\
  \intertext{\qquad{}由电荷平衡，}
  \phantom{\Rightarrow}
   & \;                                                        & \frac{1}{T_{\mathrm{s}}}\int_0^{T_{\mathrm{s}}}i_{\mathrm{C}}(t)\,\mathrm{d}t
   & \approx-D\frac{V_{\mathrm{out}}}{R}
  +D'\left(I_{\mathrm{L}}-\frac{V_{\mathrm{out}}}{R}\right)=0,
  \label{equ0210}                                                                                                                              \\
  \Rightarrow
   & \;                                                        & V_{\mathrm{out}}
   & \approx RD'I_{\mathrm{L}},
  \label{equ0211}                                                                                                                              \\
  \Rightarrow
   & \;                                                        & I_{\mathrm{L}}
   & \approx\frac{V_{\mathrm{out}}}{RD'}
  \approx\frac{V_{\mathrm{in}}}{R(D')^2+R_{\mathrm{L}}}.
  \label{equ0212}
\end{alignat}

当 $R_{\mathrm{L}}=0$ 时，以上结果退化为前章的理想 Boost 关系；对于给定的 $D$ 和负载 $R$，电感铜损使变换比降低。

\begin{alignat}{2}

\examplesubheading{建立等效电路模型}

回到前文所述模型，仅考虑直流稳态时，电感伏秒平衡关系可以写成输入侧等效回路的 KVL 方程，电容电荷平衡关系可以写成输出侧等效回路的 KCL 方程：
\begin{equation}
  \left\{
  \begin{aligned}
     & V_{\mathrm{in}}-I_{\mathrm{L}}R_{\mathrm{L}}-D'V_{\mathrm{out}}=0, \\
     & D'I_{\mathrm{L}}-\frac{V_{\mathrm{out}}}{R}=0.
  \end{aligned}
  \right.
  \label{equ0219}
\end{equation}

在该直流模型中，Boost 的输入平均电流为 $I_{\mathrm{in}}=I_{\mathrm{L}}$，输出平均电流为 $I_{\mathrm{out}}=D'I_{\mathrm{L}}$；变压器原、副边的比例为 $D':1$。

以 Buck 电路为例，其中依旧有 $V_{\mathrm{C}}=V_{\mathrm{out}}$。计及电感铜损后，省略类似推导过程，有
\begin{equation}
  \left\{
  \begin{aligned}
    \langle v_{\mathrm{L}}\rangle & =DV_{\mathrm{in}}-I_{\mathrm{L}}R_{\mathrm{L}}-V_{\mathrm{C}}=0, \\
    \langle i_{\mathrm{C}}\rangle & =I_{\mathrm{L}}-\frac{V_{\mathrm{out}}}{R}=0.
  \end{aligned}
  \right.
  \label{equ0221}
\end{equation}

对于给定的 $D$、$V_{\mathrm{in}}$、$R$ 和 $R_{\mathrm{L}}$，上述两式可直接解得
\begin{alignat}{2}
  \cdots\Rightarrow
   & \;                                                           & V_{\mathrm{out}}
   & \approx\frac{D}{1+\dfrac{R_{\mathrm{L}}}{R}}V_{\mathrm{in}},
  \label{equ0222}                                                                    \\
  \cdots\Rightarrow
   & \;                                                           & I_{\mathrm{L}}
   & \approx\frac{D}{R+R_{\mathrm{L}}}V_{\mathrm{in}}.
  \label{equ0223}
\end{alignat}

若要建立完整的输入、输出端口模型，并计算输入功率和效率，还需对左侧受控源回路进行分析，小纹波近似条件下直接可得：
\begin{equation}
  I_{\mathrm{in}}
  =\frac{1}{T_{\mathrm{s}}}\int_0^{T_{\mathrm{s}}}i_{\mathrm{in}}(t)\,\mathrm{d}t
  \approx DI_{\mathrm{L}}.
  \label{equ0224}
\end{equation}

\examplesubheading{建立等效电路模型}

为简化后续表达，记平均等效串联电阻为
\begin{equation}
  R_{\Sigma}=R_{\mathrm{L}}+DR_{\mathrm{on}}+D'R_{\mathrm{D}}.
  \label{equ0225}
\end{equation}

仅考虑直流稳态时，电感伏秒平衡关系可以写成输入侧等效回路的 KVL 方程，电容电荷平衡关系可以写成输出侧等效回路的 KCL 方程：
\begin{equation}
  \left\{
  \begin{aligned}
     & V_{\mathrm{in}}-I_{\mathrm{L}}R_{\Sigma}
    -D'V_{\mathrm{D}}-D'V_{\mathrm{out}}=0,           \\
     & D'I_{\mathrm{L}}-\frac{V_{\mathrm{out}}}{R}=0.
  \end{aligned}
  \right.
  \label{equ0226}
\end{equation}

\examplesubheading{效率与高效率条件}

在该直流平均模型中，
\begin{alignat}{2}
  \phantom{\Rightarrow}
   & \;                                                             & P_{\mathrm{in}}
   & =V_{\mathrm{in}}I_{\mathrm{in}}=V_{\mathrm{in}}I_{\mathrm{L}},
  \label{equ0228}                                                                      \\
  \phantom{\Rightarrow}
   & \;                                                             & P_{\mathrm{out}}
   & =V_{\mathrm{out}}I_{\mathrm{out}}
  \approx V_{\mathrm{out}}D'I_{\mathrm{L}}.
  \label{equ0229}                                                                      \\
  \intertext{\qquad{}因此}
  \phantom{\Rightarrow}
   & \;                                                             & \eta
   & =\frac{P_{\mathrm{out}}}{P_{\mathrm{in}}}
  \approx D'\frac{V_{\mathrm{out}}}{V_{\mathrm{in}}},
  \label{equ0230}                                                                      \\
  \Rightarrow
   & \;                                                             & \eta
   & \approx\frac{1-\dfrac{D'V_{\mathrm{D}}}{V_{\mathrm{in}}}}
  {1+\dfrac{R_{\Sigma}}{R(D')^2}}=\frac{V_{\mathrm{in}}-D'V_{\mathrm{D}}}{V_{\mathrm{in}}}
  \frac{(D')^2R}{(D')^2R+R_{\Sigma}}..
  \label{equ0231}
\end{alignat}

第一个因子对应二极管固定压降的影响；第二个因子中，$(D')^2R$ 是折算至原边的负载电阻，$R_{\Sigma}$ 是原边的总等效串联电阻。

效率较高时，应满足
\begin{equation}
  \frac{V_{\mathrm{in}}}{D'}\gg V_{\mathrm{D}},
  \qquad
  (D')^2R\gg R_{\Sigma}.
  \label{equ0232}
\end{equation}

\examplesubheading{模型准确性}

用均方根值（RMS），或称有效值估算电阻的功率损耗，对比纯直流与含纹波两种情况的影响。其中，
\begin{equation}
  I_{\mathrm{rms}}
  =\sqrt{\frac{1}{T}\int_{t_0}^{t_0+T}i^2(t)\,\mathrm{d}t}.
  \label{equ0233}
\end{equation}

电阻损耗应按
\begin{equation}
  P_R=RI_{\mathrm{rms}}^2
  =R\left(I_{\mathrm{avg}}^2+I_{\mathrm{ripple,rms}}^2\right)
  \label{equ0234}
\end{equation}
计算，其中 $I_{\mathrm{ripple,rms}}$ 为相对于平均值的电流纹波有效值。

小纹波近似下，电感铜损约为 $I_{\mathrm{L}}^2R_{\mathrm{L}}$，MOSFET 通态电阻损耗约为 $DI_{\mathrm{L}}^2R_{\mathrm{on}}$，二极管通态电阻损耗约为 $D'I_{\mathrm{L}}^2R_{\mathrm{D}}$，固定压降损耗约为 $D'V_{\mathrm{D}}I_{\mathrm{L}}$。因此，总损耗可写为
\begin{equation}
  P_{\mathrm{loss}}\approx I_{\mathrm{L}}^2R_{\Sigma}
  +D'V_{\mathrm{D}}I_{\mathrm{L}}.
  \label{equ0235}
\end{equation}
当电感电流纹波较大时，上述平均模型会遗漏纹波对有效值的贡献，损耗估算的准确性下降；同时上述模型尚未计入器件开通、关断过程中的开关损耗；进入 DCM 后需增加电感电流保持为零的第三个工作状态，并重新建立状态方程。

由于附加参数的引入，电路分析变得更加复杂，但方程的结构基本不变，结果可以依靠计算机进行求解，因此实际上建立较准确模型的关键一步应当是研究器件并掌握器件的每种主要损耗。下一章就重点关注器件本身的特性，并从器件中提取出电路模型。
