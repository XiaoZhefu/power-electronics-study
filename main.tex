\chapter{断续模式（DCM）}

\section{起源}

\exampletitle[exa0401]{Buck 变换器的断续模式}

以 Buck 为例，连续导通模式下有
\begin{align}
  I_{\mathrm{L}}=I_{\mathrm{o}}=\frac{V_{\mathrm{o}}}{R}
                        & =\frac{DV_{\mathrm{in}}}{R}, \label{equ0401}  \\
  \Delta i_{\mathrm{L}} & =\frac{V_{\mathrm{in}}DD'T_{\mathrm{s}}}{2L}.
  \label{equ0402}
\end{align}
抛去 $V_{\mathrm{in}}$，$I_{\mathrm{L}}$ 只取决于 $R$（负载）；
$\Delta i_{\mathrm{L}}$ 取决于 $D$、$T_{\mathrm{s}}$ 与 $L$ 三者。
当 $R$ 变大，$I_{\mathrm{L}}$ 减小，相对纹波有时会很大，超过 $I_{\mathrm{L}}$。
\reviewnote{“只取决于 $R$”还需固定占空比 $D$；此处沿用前文的纹波峰值定义，
  $\Delta i_{\mathrm{L}}$ 为峰峰值的一半。}

\section{轻载与重载}

恒压下，$P=V^2/R$，故 $R$ 越小，$P$ 越大，为重载；反之为轻载。
恒流下，$P=I^2R$，故 $R$ 越大，$P$ 越大，为重载；反之为轻载。

边界处 $I_{\mathrm{L}}=\Delta i_{\mathrm{L,pk}}$，即
\begin{equation}
  \frac{DV_{\mathrm{in}}}{R}=\frac{DD'T_{\mathrm{s}}V_{\mathrm{in}}}{2L}.
  \label{equ0403}
\end{equation}
当
\begin{equation}
  \frac{DV_{\mathrm{in}}}{R}<\frac{DD'T_{\mathrm{s}}V_{\mathrm{in}}}{2L}
  \quad\Rightarrow\quad \frac{2L}{RT_{\mathrm{s}}}<D'
  \label{equ0404}
\end{equation}
时，令
\begin{equation}
  K=\frac{2L}{RT_{\mathrm{s}}},\quad K_{\mathrm{crit}}(D)=D'=1-D,
  \label{equ0405}
\end{equation}
有 $K<K_{\mathrm{crit}}(D)$ 时为 DCM。
对应的临界负载电阻为
\begin{equation}
  R_{\mathrm{crit}}(D)=\frac{2L}{D'T_{\mathrm{s}}}.
  \label{equ0406}
\end{equation}
因此，$K>K_{\mathrm{crit}}(D)$ 或 $R<R_{\mathrm{crit}}(D)$ 时为 CCM；
$K<K_{\mathrm{crit}}(D)$ 或 $R>R_{\mathrm{crit}}(D)$ 时为 DCM。

\section{变换比的再分析}

储能元件稳态时，仍有
\begin{equation}
  \langle v_{\mathrm{L}}\rangle=0,\quad \langle i_{\mathrm{C}}\rangle=0.
  \label{equ0407}
\end{equation}
小纹波近似中，有 $\Delta v_{\mathrm{o}}\ll V_{\mathrm{o}}$，
故 $v_{\mathrm{o}}(t)\approx V_{\mathrm{o}}$。
但 $\Delta i_{\mathrm{L}}>I_{\mathrm{L}}$，故不能推出 $i_{\mathrm{L}}(t)\approx I_{\mathrm{L}}$。
\reviewnote{“$\Delta i_{\mathrm{L}}>I_{\mathrm{L}}$”是沿用 CCM 解判断进入 DCM 的条件。
  进入 DCM 后，应重新计算实际电流波形和平均值，不能继续直接套用 CCM 的对称纹波关系。}

\examplesubheading{状态方程}

在 $D_1T_{\mathrm{s}}$ 时间段，
\begin{equation}
  \begin{cases}
    v_{\mathrm{L}}(t)=v_{\mathrm{in}}(t)-v_{\mathrm{o}}(t)\approx V_{\mathrm{in}}-V_{\mathrm{o}}, \\
    i_{\mathrm{C}}(t)=i_{\mathrm{L}}(t)-\dfrac{v_{\mathrm{o}}(t)}{R}
    \approx i_{\mathrm{L}}(t)-\dfrac{V_{\mathrm{o}}}{R}.
  \end{cases}
  \label{equ0408}
\end{equation}
其中 $i_{\mathrm{L}}(t)$ 保留瞬时值，不作小纹波替换。
在 $D_2T_{\mathrm{s}}$ 时间段，
\begin{equation}
  \begin{cases}
    v_{\mathrm{L}}(t)=-v_{\mathrm{o}}(t)\approx-V_{\mathrm{o}}, \\
    i_{\mathrm{C}}(t)=i_{\mathrm{L}}(t)-\dfrac{v_{\mathrm{o}}(t)}{R}
    \approx i_{\mathrm{L}}(t)-\dfrac{V_{\mathrm{o}}}{R}.
  \end{cases}
  \label{equ0409}
\end{equation}
在 $D_3T_{\mathrm{s}}$ 时间段，$i_{\mathrm{L}}(t)=0$，且
\begin{equation}
  \begin{cases}
    v_{\mathrm{L}}(t)=0, \\
    i_{\mathrm{C}}(t)=-\dfrac{v_{\mathrm{o}}(t)}{R}\approx-\dfrac{V_{\mathrm{o}}}{R}.
  \end{cases}
  \label{equ0410}
\end{equation}

\examplesubheading{伏秒平衡与电荷平衡}

由电感伏秒平衡，
\begin{align}
  \langle v_{\mathrm{L}}(t)\rangle
                                  & =D_1(V_{\mathrm{in}}-V_{\mathrm{o}})+D_2(-V_{\mathrm{o}})+D_3(0)=0,
  \label{equ0411}                                                                                       \\
  \Rightarrow\quad V_{\mathrm{o}} & =V_{\mathrm{in}}\frac{D_1}{D_1+D_2}.
  \label{equ0412}
\end{align}
那么，$D_2$ 是多少？由电容电荷平衡，
\begin{equation}
  i_{\mathrm{L}}(t)=i_{\mathrm{C}}(t)+\frac{V_{\mathrm{o}}}{R},\quad
  \langle i_{\mathrm{C}}\rangle=0
  \quad\Rightarrow\quad\langle i_{\mathrm{L}}\rangle=\frac{V_{\mathrm{o}}}{R}.
  \label{equ0413}
\end{equation}
又有三角形电流波形的高度
\begin{equation}
  i_{\mathrm{L}}(D_1T_{\mathrm{s}})=\Delta i_{\mathrm{pp}}
  =\frac{V_{\mathrm{in}}-V_{\mathrm{o}}}{L}D_1T_{\mathrm{s}}.
  \label{equ0414}
\end{equation}
利用三角形面积，
\begin{align}
  \langle i_{\mathrm{L}}\rangle
   & =\frac{1}{T_{\mathrm{s}}}\frac{1}{2}\Delta i_{\mathrm{pp}}(D_1+D_2)T_{\mathrm{s}}\notag \\
   & =\frac{1}{T_{\mathrm{s}}}\frac{1}{2}
  \frac{V_{\mathrm{in}}-V_{\mathrm{o}}}{L}D_1T_{\mathrm{s}}(D_1+D_2)T_{\mathrm{s}},
  \label{equ0415}                                                                            \\
  \Rightarrow\quad\frac{V_{\mathrm{o}}}{R}
   & =\frac{D_1T_{\mathrm{s}}}{2L}(D_1+D_2)(V_{\mathrm{in}}-V_{\mathrm{o}}).
  \label{equ0416}
\end{align}

\examplesubheading{变换比}

联立
\begin{equation}
  \begin{cases}
    V_{\mathrm{o}}=V_{\mathrm{in}}\dfrac{D_1}{D_1+D_2}, \\
    \dfrac{V_{\mathrm{o}}}{R}=\dfrac{D_1T_{\mathrm{s}}}{2L}(D_1+D_2)(V_{\mathrm{in}}-V_{\mathrm{o}}),
  \end{cases}
  \label{equ0417}
\end{equation}
得
\begin{equation}
  M(D_1,K)=\frac{V_{\mathrm{o}}}{V_{\mathrm{in}}}
  =\frac{2}{1+\sqrt{1+4K/D_1^2}},\quad
  K=\frac{2L}{RT_{\mathrm{s}}}<K_{\mathrm{crit}}.
  \label{equ0418}
\end{equation}
令 $D_1=D$，合并 CCM 与 DCM 的结果，
\begin{equation}
  M(D,K)=
  \begin{cases}
    D,                            & K\ge K_{\mathrm{crit}}, \\[3pt]
    \dfrac{2}{1+\sqrt{1+4K/D^2}}, & K<K_{\mathrm{crit}}.
  \end{cases}
  \label{equ0419}
\end{equation}

\examplesubheading{伏秒平衡与电荷平衡}

由伏秒平衡，
\begin{align}
  D_1V_{\mathrm{in}}+D_2(V_{\mathrm{in}}-V_{\mathrm{o}})+D_3(0) & =0,
  \label{equ0427}                                                                                      \\
  \Rightarrow\quad V_{\mathrm{o}}                               & =\frac{D_1+D_2}{D_2}V_{\mathrm{in}}.
  \label{equ0428}
\end{align}
电感电流的峰值为
\begin{equation}
  i_{\mathrm{pk}}=\frac{V_{\mathrm{in}}}{L}D_1T_{\mathrm{s}}.
  \label{equ0429}
\end{equation}
由图 \ref{pic0410} 的二极管电流波形，利用三角形面积，
\begin{align}
  \langle i_{\mathrm{D}}\rangle
   & =\frac{1}{T_{\mathrm{s}}}\int_0^{T_{\mathrm{s}}}i_{\mathrm{D}}(t)\,\mathrm{d}t,
  \label{equ0430}                                                                    \\
  \int_0^{T_{\mathrm{s}}}i_{\mathrm{D}}(t)\,\mathrm{d}t
   & =\frac12i_{\mathrm{pk}}D_2T_{\mathrm{s}},\label{equ0431}                        \\
  \langle i_{\mathrm{D}}\rangle
   & =\frac{1}{T_{\mathrm{s}}}\left(\frac12i_{\mathrm{pk}}D_2T_{\mathrm{s}}\right)
  =\frac{V_{\mathrm{in}}D_1D_2T_{\mathrm{s}}}{2L}
  =\frac{V_{\mathrm{o}}}{R}.\label{equ0432}
\end{align}
于是，联立伏秒平衡与电荷平衡方程，
\begin{equation}
  \begin{cases}
    V_{\mathrm{o}}=\dfrac{D_1+D_2}{D_2}V_{\mathrm{in}}, \\
    \dfrac{V_{\mathrm{in}}D_1D_2T_{\mathrm{s}}}{2L}=\dfrac{V_{\mathrm{o}}}{R}.
  \end{cases}
  \label{equ0433}
\end{equation}

\examplesubheading{变换比}

首先，
\begin{equation}
  D_2=D_1\frac{V_{\mathrm{in}}}{V_{\mathrm{o}}-V_{\mathrm{in}}}.
  \label{equ0434}
\end{equation}
代入电荷平衡式，
\begin{equation}
  V_{\mathrm{o}}^2-V_{\mathrm{o}}V_{\mathrm{in}}-\frac{V_{\mathrm{in}}^2D_1^2}{K}=0,
  \label{equ0435}
\end{equation}
得
\begin{equation}
  \frac{V_{\mathrm{o}}}{V_{\mathrm{in}}}=M(D_1,K)
  =\frac{1+\sqrt{1+4D_1^2/K}}{2},\quad
  K=\frac{2L}{RT_{\mathrm{s}}}<K_{\mathrm{crit}}(D).
  \label{equ0436}
\end{equation}
令 $D_1=D$，
\begin{equation}
  M(D,K)=
  \begin{cases}
    \dfrac{1}{1-D},               & K>K_{\mathrm{crit}}, \\[4pt]
    \dfrac{1+\sqrt{1+4D^2/K}}{2}, & K<K_{\mathrm{crit}}.
  \end{cases}
  \label{equ0437}
\end{equation}
在 DCM 下，手写近似式为
\begin{equation}
  M\approx\frac12+\frac{D}{K}.
  \label{equ0438}
\end{equation}
\reviewnote{手写分母识别为 $K$。由式 \eqref{equ0436}，当 $4D^2/K\gg1$ 时应有
  $M\approx1/2+D/\sqrt K$，与截图 10 一致；原式疑漏写根号。
  式 \eqref{equ0437} 在临界点两侧取值相同，可将等号并入任一分支。}

\chapter{磁性元件模型}

\section{磁动势与磁通量}

磁动势为
\begin{equation}
  F=\int_{x_1}^{x_2}\vec H\cdot\mathrm{d}\vec l,
  \label{equ0501}
\end{equation}
类似于电压
\begin{equation}
  V=\int_{x_1}^{x_2}\vec E\cdot\mathrm{d}\vec l.
  \label{equ0502}
\end{equation}
磁通量为
\begin{equation}
  \Phi=\int_S\vec B\cdot\mathrm{d}\vec A,
  \label{equ0503}
\end{equation}
类似于电流
\begin{equation}
  I=\int_S\vec J\cdot\mathrm{d}\vec A.
  \label{equ0504}
\end{equation}
场在截面上均匀且垂直于截面时，分别有
\begin{equation}
  \Phi=BA_{\mathrm{c}},\quad I=JA_{\mathrm{c}}.
  \label{equ0505}
\end{equation}

\section{法拉第定律与安培定律}

法拉第定律为
\begin{equation}
  v(t)=\frac{\mathrm{d}\Phi(t)}{\mathrm{d}t}
  =A_{\mathrm{c}}\frac{\mathrm{d}B(t)}{\mathrm{d}t}.
  \label{equ0506}
\end{equation}
楞次定律：自抵消。
\reviewnote{式 \eqref{equ0506} 按手写保留正号，表示选定端口参考方向下的单匝绕组电压。
  若写沿指定环路方向的感应电动势，需同时核对法拉第定律的负号及磁通参考方向；
  “自抵消”宜理解为阻碍磁通的变化，而非总是抵消原磁通。}

安培定律为
\begin{equation}
  \sum I=\oint_{\mathcal L}\vec H\cdot\mathrm{d}\vec l.
  \label{equ0507}
\end{equation}
闭合路径所包围的总电流，等于磁场强度沿该闭合路径的线积分。
\reviewnote{手写说明中“对闭合曲线作线积分”的字迹有涂改，此处按该含义衔接。
  式 \eqref{equ0507} 对应忽略位移电流的磁准静态模型；电流求和应计入参考方向。}

以磁芯为例，导线中的电流 $i(t)$ 穿过磁芯窗口；图中的路径沿磁通方向绕磁芯内部一周。
磁场强度均匀时，
\begin{equation}
  F(t)=H(t)l_{\mathrm{m}}=i(t).
  \label{equ0508}
\end{equation}

未饱和时，联立得
\begin{equation}
  v(t)=\mu nA_{\mathrm{c}}\frac{\mathrm{d}H(t)}{\mathrm{d}t}
  =\frac{\mu n^2A_{\mathrm{c}}}{l_{\mathrm{m}}}\frac{\mathrm{d}i(t)}{\mathrm{d}t}
  =L\frac{\mathrm{d}i(t)}{\mathrm{d}t},
  \label{equ0516}
\end{equation}
故
\begin{equation}
  L=\frac{\mu n^2A_{\mathrm{c}}}{l_{\mathrm{m}}}.
  \label{equ0517}
\end{equation}
当 $|i|>I_{\mathrm{sat}}$，有 $B=B_{\mathrm{sat}}$，也即
\begin{equation}
  v(t)=nA_{\mathrm{c}}\frac{\mathrm{d}B_{\mathrm{sat}}}{\mathrm{d}t}\approx0,
  \label{equ0518}
\end{equation}

\reviewnote{此处保留手写表述。负向饱和时应为 $B=-B_{\mathrm{sat}}$；
  $v\approx0$ 来自完全平坦的理想饱和段。实际饱和后的增量电感仍非零，
  不能据此认为实际电感在外加电压下的端电压必然为零。}

\begin{equation}

  \mathcal R=\frac{l}{\mu A_{\mathrm{c}}},
  \label{equ0519}
\end{equation}
其中 $\mathcal R$ 为磁阻。

铁芯磁阻与气隙磁阻分别为
\begin{equation}
  \mathcal R_{\mathrm{c}}=\frac{l_{\mathrm{c}}}{\mu A_{\mathrm{c}}},\quad
  \mathcal R_{\mathrm{g}}=\frac{l_{\mathrm{g}}}{\mu_0A_{\mathrm{c}}}.
  \label{equ0523}
\end{equation}
于是
\begin{equation}
  ni=\Phi(\mathcal R_{\mathrm{c}}+\mathcal R_{\mathrm{g}}).
  \label{equ0524}
\end{equation}
沿用法拉第定律，
\begin{equation}
  v(t)=n\frac{\mathrm{d}\Phi(t)}{\mathrm{d}t}
  =\frac{n^2}{\mathcal R_{\mathrm{c}}+\mathcal R_{\mathrm{g}}}
  \frac{\mathrm{d}i(t)}{\mathrm{d}t}
  =L\frac{\mathrm{d}i(t)}{\mathrm{d}t},
  \label{equ0525}
\end{equation}
故
\begin{equation}
  L=\frac{n^2}{\mathcal R_{\mathrm{c}}+\mathcal R_{\mathrm{g}}}.
  \label{equ0526}
\end{equation}
气隙的作用：$\mu\downarrow$，但 $I_{\mathrm{sat}}\uparrow$，
即 $L\downarrow$，但 $I_{\mathrm{sat}}\uparrow$。
\reviewnote{此处的“$\mu\downarrow$”宜明确为整条磁路的等效磁导率下降，
  铁芯材料自身的磁导率并非因开气隙而直接改变。}
由 $\Phi=BA_{\mathrm{c}}$，有
\begin{equation}
  \Phi_{\mathrm{sat}}=B_{\mathrm{sat}}A_{\mathrm{c}},\quad
  I_{\mathrm{sat}}=\frac{B_{\mathrm{sat}}A_{\mathrm{c}}}{n}
  (\mathcal R_{\mathrm{c}}+\mathcal R_{\mathrm{g}}).
  \label{equ0527}
\end{equation}
气隙使电感值减小、饱和电流增大，同时使电感值对磁芯材料磁导率的依赖减弱。

\subsection{计及有限磁导率的模型}

现实中 $\mu$ 不可能无穷大，因而理想条件不能满足。
对于非零磁阻，仍有
\begin{equation}
  \Phi\mathcal R_{\mathrm{c}}=n_1i_1+n_2i_2,\quad
  v_1=n_1\frac{\mathrm{d}\Phi(t)}{\mathrm{d}t}.
  \label{equ0534}
\end{equation}
消去磁通，得
\begin{equation}
  v_1=\frac{n_1^2}{\mathcal R_{\mathrm{c}}}
  \frac{\mathrm{d}}{\mathrm{d}t}\left(i_1+\frac{n_2}{n_1}i_2\right).
  \label{equ0535}
\end{equation}
令
\begin{equation}
  L_{\mathrm{mp}}=\frac{n_1^2}{\mathcal R_{\mathrm{c}}},\quad
  i_{\mathrm{mp}}=i_1+\frac{n_2}{n_1}i_2,
  \label{equ0536}
\end{equation}
则
\begin{equation}
  v_1=L_{\mathrm{mp}}\frac{\mathrm{d}i_{\mathrm{mp}}}{\mathrm{d}t}.
  \label{equ0537}
\end{equation}
即在理想变压器的一次侧并联了一个电感 $L_{\mathrm{mp}}$。
理想变压器用于传能，$L_{\mathrm{mp}}$ 用于储能。

若 $L_{\mathrm{mp}}\to\infty$，则为理想；反之则需要考虑饱和。
\reviewnote{有限励磁电感说明需要考虑励磁电流，并不必然意味着已经饱和。
  是否饱和仍需由磁通密度、初始状态与所施加的伏秒共同判断。}

仍采用伏秒分析，
\begin{align}
  i_{\mathrm{mp}}(t) & =\frac{1}{L_{\mathrm{mp}}}\int v_1(t)\,\mathrm{d}t,
  \label{equ0538}                                                                           \\
  \lambda_1          & =\int_{t_1}^{t_2}v_1(t)\,\mathrm{d}t,\label{equ0539}                 \\
  B(t)               & =\frac{1}{n_1A_{\mathrm{c}}}\int v_1(t)\,\mathrm{d}t.\label{equ0540}
\end{align}
避免积分项过大，造成饱和。
\reviewnote{手写此处的衔接文字有涂改，按“仍采用伏秒分析”整理。
  不定积分应包含初始值；式 \eqref{equ0539} 的定积分一般等于磁链变化量
  $\lambda_1(t_2)-\lambda_1(t_1)$，只有初始磁链取零时才能直接写为终值 $\lambda_1$。}

\section{耦合电感模型}

\subsection{强耦合电感}

强耦合时，不考虑漏磁。其物理结构与变压器相似，模型电路图一致，
分别如图 \ref{pic0509a} 和图 \ref{pic0511} 所示。
将 $L_{\mathrm{mp}}$ 视为实际储能电感。

区别在于：$i_1$、$i_2$ 是否同时存在（变压器同时存在）。
决定电感量：$L_{\mathrm{mp}}$ 与 turn ratio（？）（下文有小组讨论）。
两个独立变量为 $L_{\mathrm{mp}}$ 与匝数比。
\reviewnote{手写对 turn ratio 留有问号，此处参照截图 21 识别为“匝数比”。
  “电流是否同时存在”可用于描述某些具体电路的工作时序，但不宜作为变压器与耦合电感
  的普遍定义；实际还需结合励磁储能是否为设计目的及其工作方式。手写提及的小组讨论内容未在本次图片中出现。}

与普通电感的区别：结构上，普通电感有一对接口，强耦合电感至少有两对接口。
这意味着普通电感类似一个只有一个口的水槽，“水”即能量从同一个口出入；
要求不能饱和，稳态有伏秒平衡。
而强耦合电感则有两个口，一进一出。相比而言，可以认为多个接口共享磁芯能量，可用调控平衡。
\reviewnote{“可用调控平衡”按手写暂录，原字迹需进一步确认。
  两个端口并不总是一进一出，传能方向和时序由电路决定；周期稳态下各绕组的磁链仍须复位，
  不应理解为强耦合电感不再需要伏秒平衡。}

\subsection{弱耦合电感（无磁芯）}

弱耦合电感的电压、电流关系为
\begin{equation}
  \begin{cases}
    v_1=\mathrm{j}\omega L_1i_1+\mathrm{j}\omega M_{12}i_2, \\
    v_2=\mathrm{j}\omega M_{12}i_1+\mathrm{j}\omega L_2i_2,
  \end{cases}
  \label{equ0541}
\end{equation}
等价地，
\begin{equation}
  \begin{bmatrix}v_1\\v_2\end{bmatrix}
  =\mathrm{j}\omega
  \begin{bmatrix}L_1&M_{12}\\M_{12}&L_2\end{bmatrix}
  \begin{bmatrix}i_1\\i_2\end{bmatrix}.
  \label{equ0542}
\end{equation}
\reviewnote{标题中的“无磁芯”按手写保留，但弱耦合并不必然没有磁芯。
  式 \eqref{equ0541}、\eqref{equ0542} 含 $\mathrm{j}\omega$，属于正弦稳态相量关系，
  其中 $v_1,v_2,i_1,i_2$ 的符号目前沿用手写；后续宜明确其为相量，并核对同名端及电流参考方向。}

\begin{align}
  \frac{V_1}{V_{\mathrm{in}}} & =M_1(D),\label{equ0603} \\
  \frac{V_{\mathrm{o}}}{V_1}  & =M_2(D).\label{equ0604}
\end{align}
因此，总变换比为
\begin{equation}
  \frac{V_{\mathrm{o}}}{V_{\mathrm{in}}}=M(D)=M_1(D)M_2(D).
  \label{equ0605}
\end{equation}

\section{差分连接}

两个变换器共用输入电源，负载接在两个输出端之间。
变换器 1 由 $D$ 控制，变换器 2 由 $D'$ 控制。手写框图中的关系为
\begin{align}
  V_1 & =M(D)V_{\mathrm{in}},\label{equ0610}  \\
  V_1 & =M(D')V_{\mathrm{in}}.\label{equ0611}
\end{align}
\reviewnote{下方变换器框内的电压下标暂识别为 $1$，按原稿保留在式 \eqref{equ0611} 和框图中；

\begin{equation}

\end{equation}

接位置 1 时，
\begin{equation}
  v_{\mathrm{L}}=V_{ab}.
  \label{equ0619}
\end{equation}
接位置 2 时，
\begin{equation}
  v_{\mathrm{L}}=V_{cb}.
  \label{equ0620}
\end{equation}
由电感伏秒平衡，
\begin{equation}
  DV_{ab}+D'V_{cb}=0.
  \label{equ0621}
\end{equation}

\subsubsection{接法一：$a$--$A$、$b$--$B$、$c$--$C$}

\begin{equation}
  \begin{cases}
    V_{ab}=V_{AB}=V_{\mathrm{in}}-V_{\mathrm{o}}, \\
    V_{cb}=V_{CB}=-V_{\mathrm{o}}.
  \end{cases}
  \label{equ0622}
\end{equation}
\begin{align}
  \Rightarrow\quad & D(V_{\mathrm{in}}-V_{\mathrm{o}})+D'(-V_{\mathrm{o}})=0,
  \label{equ0623}                                                             \\
  \Rightarrow\quad & V_{\mathrm{o}}=DV_{\mathrm{in}}.\label{equ0624}
\end{align}
此时为 Buck 变换器。

\subsubsection{接法二：$a$--$C$、$b$--$A$、$c$--$B$}

\begin{equation}
  \begin{cases}
    V_{ab}=V_{CA}=-V_{\mathrm{in}}, \\
    V_{cb}=V_{BA}=V_{\mathrm{o}}-V_{\mathrm{in}}.
  \end{cases}
  \label{equ0625}
\end{equation}
\begin{align}
  \Rightarrow\quad & D(-V_{\mathrm{in}})+D'(V_{\mathrm{o}}-V_{\mathrm{in}})=0,
  \label{equ0626}                                                               \\
  \Rightarrow\quad & V_{\mathrm{o}}=\frac{1}{D'}V_{\mathrm{in}}.\label{equ0627}
\end{align}
此时为 Boost 变换器。
\reviewnote{手写式 \eqref{equ0627} 的分母有涂改，此处按最终字迹识别为 $D'$；
  这一读法也与式 \eqref{equ0626} 一致。}

\subsubsection{接法三：$a$--$A$、$b$--$C$、$c$--$B$}

\begin{equation}
  \begin{cases}
    V_{ab}=V_{AC}=V_{\mathrm{in}}, \\
    V_{cb}=V_{BC}=V_{\mathrm{o}}.
  \end{cases}
  \label{equ0628}
\end{equation}
\begin{align}
  \Rightarrow\quad & DV_{\mathrm{in}}+D'V_{\mathrm{o}}=0,
  \label{equ0629}                                                                \\
  \Rightarrow\quad & V_{\mathrm{o}}=-\frac{D}{D'}V_{\mathrm{in}}.\label{equ0630}
\end{align}
此时为 Buck--Boost 变换器。
\reviewnote{上述推导以 $D'=1-D$ 为前提，电压变换关系采用理想器件、周期稳态和连续导通的两状态模型。
  “旋转”表示三端单元与外部端口重新配接，电压下标与参考极性需随接法一并核对。}
