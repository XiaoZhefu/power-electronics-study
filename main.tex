\chapter{稳态等效模型}

\reviewnote{原稿章标题在“稳态等效模型”之后有划改，后半部分未能明确辨认，暂保留能够确认的标题。原稿输出记号混用 $V_{\mathrm{o}}$ 与 $V_{\mathrm{out}}$，个别手写下标 $o/0$ 难以区分，以下统一用 $V_{\mathrm{out}}$、$I_{\mathrm{out}}$ 表示输出平均电压和电流。}

\section{引入模型的必要性}

即使是理想电路，在瞬态过程中，滤波器的储能元器件也可能吸收或释放能量，因此输入功率与输出功率可能不相等，即 $P_{\mathrm{in}}\neq P_{\mathrm{out}}$。

\reviewnote{这里的功率不等主要针对瞬态的能量变化。若讨论瞬时功率，建议写为 $p_{\mathrm{in}}(t)\neq p_{\mathrm{out}}(t)$；理想周期稳态下，一个周期内净储能变化为零，输入与输出的周期平均功率相等。暂保留原稿的 $P$ 记号。}

\section{理想稳态模型}

理想稳态下，输入功率与输出功率相等：
\begin{align}
  P_{\mathrm{in}}                & =P_{\mathrm{out}},                 \\
  V_{\mathrm{in}}I_{\mathrm{in}} & =V_{\mathrm{out}}I_{\mathrm{out}}.
\end{align}
若直流电压变换比为 $M(D)$，则
\begin{align}
  V_{\mathrm{out}} & =M(D)V_{\mathrm{in}},  \\
  I_{\mathrm{in}}  & =M(D)I_{\mathrm{out}}.
\end{align}

\reviewnote{上述 $VI$ 写法采用直流端口模型。若端口电压、电流含有明显纹波，平均功率应由 $\langle v(t)i(t)\rangle$ 求得，通常不能直接用两个平均值的乘积代替。}

上述端口关系也可以用变比为 $1:M(D)$ 的直流“变压器”表示，$D$ 为控制输入。

\begin{equation}
  V_{\mathrm{out}}
  =M(D)V_1\frac{R}{R+M^2(D)R_1}.
\end{equation}

\subsection{状态方程与小纹波近似}

状态 1 时，
\begin{equation}
  \left\{
  \begin{aligned}
    v_{\mathrm{L}}(t) & =V_{\mathrm{in}}-i_{\mathrm{L}}(t)R_{\mathrm{L}}, \\
    i_{\mathrm{C}}(t) & =-\frac{v_{\mathrm{out}}(t)}{R}.
  \end{aligned}
  \right.
\end{equation}
作小纹波近似，有
\begin{equation}
  \left\{
  \begin{aligned}
    v_{\mathrm{L}}(t) & \approx V_{\mathrm{in}}-I_{\mathrm{L}}R_{\mathrm{L}}, \\
    i_{\mathrm{C}}(t) & \approx-\frac{V_{\mathrm{out}}}{R}.
  \end{aligned}
  \right.
\end{equation}

状态 2 时，
\begin{equation}
  \left\{
  \begin{aligned}
    v_{\mathrm{L}}(t) & =V_{\mathrm{in}}-i_{\mathrm{L}}(t)R_{\mathrm{L}}-v_{\mathrm{out}}(t), \\
    i_{\mathrm{C}}(t) & =i_{\mathrm{L}}(t)-\frac{v_{\mathrm{out}}(t)}{R}.
  \end{aligned}
  \right.
\end{equation}
作小纹波近似，有
\begin{equation}
  \left\{
  \begin{aligned}
    v_{\mathrm{L}}(t) & \approx V_{\mathrm{in}}-I_{\mathrm{L}}R_{\mathrm{L}}-V_{\mathrm{out}}, \\
    i_{\mathrm{C}}(t) & \approx I_{\mathrm{L}}-\frac{V_{\mathrm{out}}}{R}.
  \end{aligned}
  \right.
\end{equation}

\reviewnote{原稿状态 2 近似式中输出电压的时间记号不够清晰，暂按此处取平均量的上下文整理为 $V_{\mathrm{out}}$，待核对。以下推导沿用前章的 CCM、周期稳态和小纹波近似条件。}

\subsection{电感伏秒平衡}

由伏秒平衡，
\begin{align}
  \langle v_{\mathrm{L}}\rangle
   & =\frac{1}{T_{\mathrm{s}}}\int_0^{T_{\mathrm{s}}}v_{\mathrm{L}}(t)\,\mathrm{d}t \\
   & \approx D(V_{\mathrm{in}}-I_{\mathrm{L}}R_{\mathrm{L}})
  +D'(V_{\mathrm{in}}-I_{\mathrm{L}}R_{\mathrm{L}}-V_{\mathrm{out}})=0.
\end{align}
因此
\begin{equation}
  V_{\mathrm{in}}-I_{\mathrm{L}}R_{\mathrm{L}}-D'V_{\mathrm{out}}=0.
\end{equation}

\subsection{电容电荷平衡与变换比}

由电荷平衡，
\begin{align}
  \langle i_{\mathrm{C}}\rangle
   & =\frac{1}{T_{\mathrm{s}}}\int_0^{T_{\mathrm{s}}}i_{\mathrm{C}}(t)\,\mathrm{d}t \\
   & \approx-D\frac{V_{\mathrm{out}}}{R}
  +D'\left(I_{\mathrm{L}}-\frac{V_{\mathrm{out}}}{R}\right)=0.
\end{align}
因此
\begin{equation}
  D'I_{\mathrm{L}}-\frac{V_{\mathrm{out}}}{R}=0.
\end{equation}
联立两个平衡关系，得
\begin{equation}
  \frac{V_{\mathrm{out}}}{V_{\mathrm{in}}}
  =\frac{1}{D'\left(1+\dfrac{R_{\mathrm{L}}}{(D')^2R}\right)}.
\end{equation}

\subsection{效率与高效率条件}

在该直流平均模型中，
\begin{align}
  P_{\mathrm{in}}
   & =V_{\mathrm{in}}I_{\mathrm{in}}=V_{\mathrm{in}}I_{\mathrm{L}},      \\
  P_{\mathrm{out}}
   & =V_{\mathrm{out}}I_{\mathrm{out}}=V_{\mathrm{out}}D'I_{\mathrm{L}}.
\end{align}
因此
\begin{align}
  \eta
   & =D'\frac{V_{\mathrm{out}}}{V_{\mathrm{in}}}         \\
   & =\frac{1-\dfrac{D'V_{\mathrm{D}}}{V_{\mathrm{in}}}}
  {1+\dfrac{R_{\mathrm{L}}+DR_{\mathrm{on}}+D'R_{\mathrm{D}}}{(D')^2R}}.
\end{align}

效率较高时，原稿给出的条件为
\begin{equation}
  \frac{V_{\mathrm{in}}}{D'}\gg V_{\mathrm{D}},
  \qquad
  (D')^2R\gg R_{\mathrm{L}}+DR_{\mathrm{on}}+D'R_{\mathrm{D}}.
\end{equation}
原稿随后写为
\begin{equation}
  \eta\approx
  \frac{V_{\mathrm{in}}-D'V_{\mathrm{D}}}{V_{\mathrm{in}}}
  \frac{(D')^2R}{(D')^2R+R_{\mathrm{L}}+DR_{\mathrm{on}}+D'R_{\mathrm{D}}},
\end{equation}
并在第二个因子下标注了“折算至原边”。

\reviewnote{本行右侧与上一条效率表达式代数等价，尚未实际使用“高效率”条件作近似，原稿的约等号暂保留。若需要一阶近似，可进一步写为 $\eta\approx1-D'V_{\mathrm{D}}/V_{\mathrm{in}}-(R_{\mathrm{L}}+DR_{\mathrm{on}}+D'R_{\mathrm{D}})/[(D')^2R]$，并忽略小量乘积。“折算至原边”的箭头指向范围不够明确，需结合后续等效电路图核对。}

\textbf{疑问：}准确性如何？

用均方根值（有效值，即 RMS）估算电阻的功率损耗，对比纯直流与含纹波两种情况的影响。原稿指出，当 $I_{\mathrm{L}}$ 的纹波较大时，该模型的准确性不足；进入 DCM 时问题更明显。

\reviewnote{电阻损耗应按 $P_R=RI_{\mathrm{rms}}^2$ 计算，直接以平均电流平方代替会忽略纹波的均方值。上述模型还未计入开关损耗等因素；进入 DCM 后应加入零电感电流区间，不能直接沿用这里只有两个状态的 CCM 模型。}
