\DefineEquation{equ0307}{
  \begin{align}
     & \left\{
    \begin{aligned}
      v_{\mathrm{L}}(t) & =V_{\mathrm{in}}-v_{\mathrm{out}}(t)
      \approx V_{\mathrm{in}}-V_{\mathrm{out}},                            \\
      i_{\mathrm{C}}(t) & =i_{\mathrm{L}}(t)-\frac{v_{\mathrm{out}}(t)}{R}
      \approx I_{\mathrm{L}}-\frac{V_{\mathrm{out}}}{R}.
    \end{aligned}
    \right.
    \label{equ0307} \\
    \intertext{\qquad{}接状态 2 时，}
     & \left\{
    \begin{aligned}
      v_{\mathrm{L}}(t) & =-v_{\mathrm{out}}(t)
      \approx -V_{\mathrm{out}},                                           \\
      i_{\mathrm{C}}(t) & =i_{\mathrm{L}}(t)-\frac{v_{\mathrm{out}}(t)}{R}
      \approx I_{\mathrm{L}}-\frac{V_{\mathrm{out}}}{R}.
    \end{aligned}
    \right.
    \label{equ0308}
  \end{align}
}

\DefineEquation{equ0309}{
  \begin{alignat}{2}
    \phantom{\Rightarrow}
     & \;                                                                & \frac{1}{T_{\mathrm{s}}}\int_0^{T_{\mathrm{s}}}v_{\mathrm{L}}(t)\,\mathrm{d}t
     & \approx D(V_{\mathrm{in}}-V_{\mathrm{out}})-D'V_{\mathrm{out}}=0,
    \label{equ0309}                                                                                                                                      \\
    \Rightarrow
     & \;                                                                & V_{\mathrm{out}}
     & \approx DV_{\mathrm{in}},
    \label{equ0310}                                                                                                                                      \\
    \Rightarrow
     & \;                                                                & M(D)
     & =\frac{V_{\mathrm{out}}}{V_{\mathrm{in}}}
    \approx D=D_{\mathrm{c}}-f_{\mathrm{s}}t_{\mathrm{rr}}.
    \label{equ0311}                                                                                                                                      \\
    \intertext{\qquad{}由电荷平衡，}
    \phantom{\Rightarrow}
     & \;                                                                & \frac{1}{T_{\mathrm{s}}}\int_0^{T_{\mathrm{s}}}i_{\mathrm{C}}(t)\,\mathrm{d}t
     & \approx D\left(I_{\mathrm{L}}-\frac{V_{\mathrm{out}}}{R}\right)
    +D'\left(I_{\mathrm{L}}-\frac{V_{\mathrm{out}}}{R}\right)=0,
    \label{equ0312} \displaybreak[1]                                                                                                                     \\
    \Rightarrow
     & \;                                                                & V_{\mathrm{out}}
     & \approx RI_{\mathrm{L}},
    \label{equ0313}                                                                                                                                      \\
    \Rightarrow
     & \;                                                                & I_{\mathrm{L}}
     & \approx\frac{V_{\mathrm{out}}}{R}
    \approx\frac{DV_{\mathrm{in}}}{R}.
    \label{equ0314}
  \end{alignat}
}

\DefineEquation{equ0315}{
  \begin{equation}
    \begin{aligned}
      I_{\mathrm{in}}
       & =\frac{1}{T_{\mathrm{s}}}\int_0^{T_{\mathrm{s}}}i_{\mathrm{Q}}(t)\,\mathrm{d}t                            \\
       & \approx\frac{DT_{\mathrm{s}}I_{\mathrm{L}}+t_{\mathrm{rr}}I_{\mathrm{L}}+Q_{\mathrm{rr}}}{T_{\mathrm{s}}} \\
       & =DI_{\mathrm{L}}+\frac{t_{\mathrm{rr}}}{T_{\mathrm{s}}}I_{\mathrm{L}}
      +\frac{Q_{\mathrm{rr}}}{T_{\mathrm{s}}}.
    \end{aligned}
    \label{equ0315}
  \end{equation}
}

\DefineEquation{equ0316}{
  \begin{equation}
    \begin{aligned}
      \Delta i_{\mathrm{L,pp}}
       & \approx\frac{V_{\mathrm{in}}-V_{\mathrm{out}}}{L}DT_{\mathrm{s}}
      \approx\frac{V_{\mathrm{in}}DD'T_{\mathrm{s}}}{L},                  \\
      \Delta v_{\mathrm{out,pp}}
       & \approx\frac{\Delta i_{\mathrm{L,pp}}T_{\mathrm{s}}}{8C}.
    \end{aligned}
    \label{equ0316}
  \end{equation}
}

\DefineEquation{equ0317}{
  \begin{align}
    P_{\mathrm{in}}

    +\frac{Q_{\mathrm{rr}}}{T_{\mathrm{s}}}\right),\label{equ0317}            \\
    P_{\mathrm{out}} & \approx V_{\mathrm{out}}I_{\mathrm{L}},\label{equ0318} \\
    P_{\mathrm{sw}}  & =P_{\mathrm{in}}-P_{\mathrm{out}}
    \approx V_{\mathrm{in}}\left(\frac{t_{\mathrm{rr}}}{T_{\mathrm{s}}}I_{\mathrm{L}}
    +\frac{Q_{\mathrm{rr}}}{T_{\mathrm{s}}}\right).\label{equ0319}
  \end{align}
}

\DefineEquation{equ0320}{
